type DemoScenario = typeof DEMO_SCENARIOS[number];

interface DemoReadiness {
  synthetic_ready: boolean;
  live_ready: boolean | null;
  live_ready_reasons: string[];
  host: { enrolled: boolean; hostname: string | null; os_info: string | null } | null;
  flow_collector_active: boolean | null;
}

// Helper: find a demo scenario by id (narrowed from the imported ScenariosResponse type)
function findDemoScenario(id: DemoScenario["id"]): DemoScenario {
  const result = DEMO_SCENARIOS.find((s) => s.id === id);
  if (!result) throw new Error(`Unknown demo scenario: ${id}`);
  return result;
}

              {isRunning ? "Running…" : `Run "${scenario.name}"`}
            </button>
          </div>
  )}

          {isRunning && (
            <div className="mt-3 text-sm text-muted">
              Simulation in progress — check backend logs for detection output.
            </div>
          </div>
  )}
        </div>

        {/* Live readiness detail */}
        {hostId && readiness.live_ready_reasons.length > 0 && (
          <details className="mt-3">
            <summary className="text-xs text-caption font-medium text-muted capitalize">Live-mode readiness</summary>
            <ul className="text-[10px] text-muted space-y-1">
              {readiness.live_ready_reasons.map((r) => (
                <li key={r}>{r}</li>
              ))}
            </ul>
          </details>
        </div>
  )}
      </div>
    );
  }, [enabled, selectedScenario, hostId, readiness, runMutation]);

  // Since Demo Center manages its own state, we render a simplified view
  // without the AttackLab tab machinery.
  const demoContent = useMemo(() => {
    if (!selectedScenario) {
      return (
        <EmptyState
          title="Select a demonstration scenario"
          description="Choose a scenario from the catalog to learn what it demonstrates and run it."
        />
      );
    }
    return presetsContent;
  }, [selectedScenario]);

  return (
    <div className="space-y-6">
      <AttackLabHeader enabled={user?.role?.name !== "viewer"} />

      {user?.role?.name !== "viewer" && (
        <LiveAttackBanner
          running={isRunning}
          hostId={hostId ? hostId.slice(0, 8) : ""}
          hostName={hostId ? `Demo Host ${hostId.slice(0, 8)}` : "Demo Host"}
          scenarioName={selectedScenario?.name}
          expectedEvents={expectedEvents}
          runStartedAt={runStartedAt}
        />
      </div>
  )}

      {user?.role?.name !== "viewer" && (
        <>
          <SimulationLiveProgress
            running={isRunning}
            scenarioName={selectedScenario?.name}
            expectedEvents={expectedEvents}
            runStartedAt={runStartedAt}
          />

          <div className="grid lg:grid-cols-[minmax(0,1.4fr)_minmax(280px,1fr)] gap-6">
            <div className="space-y-6">
              {demoContent}

              <HistoryContent selectedRunId={selectedRunId} />
            </div>
          </div>

          <InvestigationGuide />

          <SimulationResults
            result={lastResult}
            onRunAgain={() => {
              setLastResult(null);
              setSelectedRunId(null);
            }}
            onReplay={undefined}
            onReset={undefined}
            replayPending={false}
            resetPending={false}
          />

          <SimulationFeed running={isRunning} />

          {hostId && (
            <RawLogTail hostId={hostId} hostName={`Demo Host ${hostId.slice(0, 8)}`} running={isRunning} />
          </div>
  )}

          <ConfirmDialog
            open={purgeConfirmOpen}
            onClose={() => setPurgeConfirmOpen(false)}
            onConfirm={() => {
              if (confirm("Remove all demo data?")) {
                // purge via mutation if available
                runMutation?.mutate?.(hostId || "");
              }
              setPurgeConfirmOpen(false);
            }}
            title="Purge simulation data"
            description="Remove all simulated events, timelines, open alerts, and run history? This cannot be unrun."
            confirmLabel="Purge all"
            danger
            loading={runMutation?.isPending}
          />
        </>
      ) : (
        <EmptyState
          title="Demo Center unavailable"
          description="Admin or analyst access required, or simulation disabled in this environment."
          action="/login"
          actionLabel="Log in"
        />
      </div>
  )}
    </div>
  );
}

export function DemoCenter() {
  return (
    <div>
      <DemoCenterPage />
    </div>
  );
}
